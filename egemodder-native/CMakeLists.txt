cmake_minimum_required(VERSION 3.18)
project(egemodder_native LANGUAGES CXX)

add_library(egemodder SHARED src/egemodder.cpp)
target_compile_features(egemodder PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(egemodder PRIVATE -Wall -Wextra -Werror)
endif()
target_link_libraries(egemodder PRIVATE dl)

if (NOT ANDROID)
  include(CTest)
  if (BUILD_TESTING)
    add_library(fake_il2cpp SHARED tests/fake_il2cpp.cpp)
    set_target_properties(fake_il2cpp PROPERTIES OUTPUT_NAME il2cpp)
    add_executable(egemodder_native_test tests/native_test.cpp)
    target_compile_features(egemodder_native_test PRIVATE cxx_std_17)
    target_link_libraries(egemodder_native_test PRIVATE egemodder fake_il2cpp)
    add_test(NAME egemodder_native_mock COMMAND egemodder_native_test)
  endif()
endif()
