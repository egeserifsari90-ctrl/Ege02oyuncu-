cmake_minimum_required(VERSION 3.18)
project(egemodder_native LANGUAGES CXX)
add_library(egemodder SHARED src/egemodder.cpp src/player_hooks.cpp)
target_compile_features(egemodder PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(egemodder PRIVATE -Wall -Wextra -Werror)
endif()
target_link_libraries(egemodder PRIVATE ${CMAKE_DL_LIBS})

if(ANDROID)
  if(NOT ANDROID_ABI STREQUAL "armeabi-v7a")
    message(FATAL_ERROR "Supplied SBA 13.0.8 game split contains armeabi-v7a only")
  endif()
  if(NOT EXISTS "${CMAKE_SOURCE_DIR}/third_party/dobby/armeabi-v7a/libdobby.a")
    message(FATAL_ERROR "Unpack existing Dobby archive first")
  endif()
  add_library(egemodder_dobby STATIC IMPORTED)
  set_target_properties(egemodder_dobby PROPERTIES IMPORTED_LOCATION
    "${CMAKE_SOURCE_DIR}/third_party/dobby/armeabi-v7a/libdobby.a")
  target_link_libraries(egemodder PRIVATE egemodder_dobby log dl)
  add_library(egemodder_main SHARED src/loader.cpp)
  set_target_properties(egemodder_main PROPERTIES OUTPUT_NAME main)
  target_link_libraries(egemodder_main PRIVATE log dl)
endif()

if(NOT ANDROID)
  include(CTest)
  if(BUILD_TESTING)
    add_library(fake_il2cpp SHARED tests/fake_il2cpp.cpp)
    set_target_properties(fake_il2cpp PROPERTIES OUTPUT_NAME il2cpp)
    add_executable(egemodder_native_test tests/native_test.cpp)
    target_compile_features(egemodder_native_test PRIVATE cxx_std_17)
    target_link_libraries(egemodder_native_test PRIVATE egemodder fake_il2cpp)
    add_test(NAME egemodder_native_mock COMMAND egemodder_native_test)
  endif()
endif()
