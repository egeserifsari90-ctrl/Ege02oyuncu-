cmake_minimum_required(VERSION 3.18)
project(SBAGOLD LANGUAGES CXX)
enable_testing()
add_executable(sbagold_patch_test tests/sbagold_patch_test.cpp)
target_include_directories(sbagold_patch_test PRIVATE include)
target_compile_features(sbagold_patch_test PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(sbagold_patch_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME sbagold_patch_behavior COMMAND sbagold_patch_test)
